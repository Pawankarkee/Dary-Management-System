```
## 💡 PART 3: SMART PLACEHOLDER AUTO-FILL SYSTEM

### **Problem Statement:**
```
CURRENT STATE:
❌ Rate field shows "45.0" on every new entry (confusing)
❌ User has to clear and retype even if value is same
❌ No visual indication of previous values
❌ Entering same values repeatedly is tedious

REQUIRED BEHAVIOR:
✅ Rate field starts BLANK (empty) on first-time farmer entry
✅ On subsequent entries for SAME farmer:
   - Quantity field shows placeholder: "Last: 18.5 L"
   - Rate field shows placeholder: "Last: रु 65/L"
   - FAT field shows placeholder: "Last: 5%"
   - SNF field shows placeholder: "Last: 8%"
✅ If user presses Enter/Tab WITHOUT typing:
   - Use placeholder value automatically
   - Move to next field
✅ If user types anything:
   - Clear placeholder
   - Use new typed value
✅ On form submit (Save button):
   - Save whatever is in fields (typed OR placeholder values)
Implementation Logic:
dart// STEP 1: Fetch last entry when farmer selected
Future<void> _onFarmerSelected(String farmerId) async {
  // Query local database
  final lastEntry = await _database.getLastMilkEntry(farmerId);
  
  if (lastEntry != null) {
    setState(() {
      // Store previous values
      _previousQuantity = lastEntry.quantity;
      _previousRate = lastEntry.rate;
      _previousFAT = lastEntry.fat;
      _previousSNF = lastEntry.snf;
      
      // Set placeholders (don't fill text controllers)
      _quantityPlaceholder = 'Last: ${lastEntry.quantity} L';
      _ratePlaceholder = 'Last: रु ${lastEntry.rate}/L';
      _fatPlaceholder = lastEntry.fat != null ? 'Last: ${lastEntry.fat}%' : null;
      _snfPlaceholder = lastEntry.snf != null ? 'Last: ${lastEntry.snf}%' : null;
    });
  } else {
    // First time entry for this farmer
    setState(() {
      _previousQuantity = null;
      _previousRate = null;
      _previousFAT = null;
      _previousSNF = null;
      _quantityPlaceholder = 'Enter quantity';
      _ratePlaceholder = 'Enter rate';
      _fatPlaceholder = 'Enter FAT value';
      _snfPlaceholder = 'Enter SNF value';
    });
  }
}

// STEP 2: TextField with smart placeholder
TextField(
  controller: _quantityController,
  focusNode: _quantityFocus,
  keyboardType: TextInputType.numberWithOptions(decimal: true),
  textInputAction: TextInputAction.next,
  decoration: InputDecoration(
    labelText: 'Quantity (Liters) *',
    hintText: _quantityPlaceholder, // Shows "Last: 18.5 L"
    hintStyle: TextStyle(
      color: Colors.blue.shade600, // Different color to show it's previous value
      fontStyle: FontStyle.italic,
      fontSize: MobileSizes.bodySmall,
    ),
  ),
  onFieldSubmitted: (_) {
    // If user didn't type anything, use placeholder value
    if (_quantityController.text.isEmpty && _previousQuantity != null) {
      _quantityController.text = _previousQuantity.toString();
    }
    // Move to next field
    FocusScope.of(context).requestFocus(_rateFocus);
  },
)

// STEP 3: On form submit, use placeholder if field empty
void _submitForm() {
  // Quantity
  double quantity;
  if (_quantityController.text.isNotEmpty) {
    quantity = double.parse(_quantityController.text);
  } else if (_previousQuantity != null) {
    quantity = _previousQuantity!; // Use placeholder value
  } else {
    // Show error: Quantity required
    return;
  }
  
  // Rate
  double rate;
  if (_rateController.text.isNotEmpty) {
    rate = double.parse(_rateController.text);
  } else if (_previousRate != null) {
    rate = _previousRate!; // Use placeholder value
  } else {
    // Show error: Rate required
    return;
  }
  
  // FAT (optional)
  double? fat;
  if (_fatController.text.isNotEmpty) {
    fat = double.parse(_fatController.text);
  } else if (_previousFAT != null) {
    fat = _previousFAT; // Use placeholder value if available
  }
  
  // SNF (optional)
  double? snf;
  if (_snfController.text.isNotEmpty) {
    snf = double.parse(_snfController.text);
  } else if (_previousSNF != null) {
    snf = _previousSNF; // Use placeholder value if available
  }
  
  // Save entry
  _saveMilkEntry(quantity, rate, fat, snf);
}
